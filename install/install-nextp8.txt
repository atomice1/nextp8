10 REM nextp8 Install Program
20 REM This program installs nextp8 from the external SD card to the internal SD card
30 POKE 23692, 255
40 RESTORE
50 PROC main()
60 PROC reset()

70 DEFPROC reset()
75 PRINT "Press any key to reset"
80 ON ERROR REG 2,2
85 NEXT #0 TO k
90 REG 2,2
95 STOP
99 ENDPROC

100 DEFPROC checkfile(filename$)
120 ON ERROR CLOSE #4: ENDPROC =0
130 CLOSE #4
140 OPEN #4, filename$
160 CLOSE #4
180 ENDPROC =1

200 DEFPROC createpath(path$)
210 ON ERROR ENDPROC
220 MKDIR path$
240 ENDPROC

250 DEFPROC createdirs()
260 ON ERROR ENDPROC
270 READ %n
280 FOR i=1 TO %n
290     READ dir$
300     PROC createpath(target$+dir$)
310 NEXT i
330 ENDPROC

350 DEFPROC main()
360 CLS
370 INVERSE 1:PRINT AT 0,0;"nextp8 Installation Program":INVERSE 0
380 PRINT AT 2,0;
400 REM Drive detection variables
410 LET source$ = ""
420 LET target$ = ""
440 REM Check for installer file on C: drive
445 PRINT "Starting install..."
450 PRINT "Checking C:..."
460 PROC checkfile("C:/install-nextp8.bas") TO result
470 IF result = 1
475     LET source$ = "C:/"
480     LET target$ = "D:/"
490     GOTO @sourcefound
500 ENDIF
520 REM Check for installer file on D: drive
530 PROC checkfile("D:/install-nextp8.bas") TO result
540 IF result = 1
550     LET source$ = "D:/"
560     LET target$ = "C:/"
570     GOTO @sourcefound
580 ENDIF
600 REM No installer file found - display error
620 PRINT "ERROR: No install files found"
630 PRINT "Please copy install-nextp8.bat to either C: or D: drive"
670 PROC reset()
680 @sourcefound
700 REM Target validation - check for TBBLUE.FW (NextZXOS firmware file)
720 PRINT "Source drive: ";:INVERSE 1:PRINT source$:INVERSE 0
730 PRINT "Target drive: ";:INVERSE 1:PRINT target$:INVERSE 0
740 PRINT "Checking target system..."
760 PROC checkfile(target$+"TBBLUE.FW") TO result
770 IF result = 0
790     PRINT "ERROR: No TBBLUE.FW found"
800     PRINT "Target drive must have NextZXOS installed"
840     PROC reset()
850 ENDIF
880 REM User confirmation
920 PRINT "Install nextp8 distribution     files?"
930 PRINT "Press Y to confirm, any other   key to abort..."
940 NEXT #0 TO k
950 IF k <> CODE "Y" AND k <> CODE "y" THEN PROC reset()
990 REM Check if already installed
1000 PROC checkfile(target$+"machines/nextp8/core.cfg") TO result
1010 IF result = 1
1030     PRINT "nextp8 is already installed on  ";target$
1040     PRINT "Press Y to confirm overwrite,   any other key to abort..."
1050     NEXT #0 TO k
1060     IF k <> CODE "Y" AND k <> CODE "y" THEN PROC reset()
1110 ENDIF
1150 REM Create target directory structure
1170 PRINT "Creating directory structure..."
1180 PROC createdirs()
1200 REM Copy files using COPY command
1220 PRINT "Installing nextp8 files..."
1230 INVERSE 1:PRINT "Copying files from ";source$;" to ";target$:INVERSE 0
1260 DATA 8

1610    PRINT filepath$

1630 NEXT i
1640 @copydone
